// Q15. Accept a 3-digit number and check whether it is an Armstrong number using arithmetic operators and if-else.

import java.util.Scanner;

public class Lab01_Q15 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q15: Armstrong Number Check (3-digit) ---");
        System.out.print("Enter a 3-digit number: ");
        int num = sc.nextInt();
        if (num < 100 || num > 999) {
            System.out.println("Please enter a valid 3-digit number.");
        } else {
            int original = num;
            int sum = 0;
            while (num > 0) {
                int digit = num % 10;
                sum += (digit * digit * digit);
                num /= 10;
            }
            if (sum == original) {
                System.out.println(original + " is an Armstrong number.");
            } else {
                System.out.println(original + " is NOT an Armstrong number.");
            }
        }
        sc.close();
    }
}



// Q16. Input salary and calculate tax based on income slabs using if-else.

public class Lab01_Q16 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q16: Income Tax Calculation ---");

        double salary = sc.nextDouble();
        double tax = 0;

        if (salary <= 250000) {
            tax = 0;
        } else if (salary <= 500000) {
            tax = (salary - 250000) * 0.05;
        } else if (salary <= 1000000) {
            tax = (250000 * 0.05) + (salary - 500000) * 0.20;
        } else {
            tax = (250000 * 0.05) + (500000 * 0.20) + (salary - 1000000) * 0.30;
        }
        System.out.println("Calculated Income Tax: ₹" + tax);
        sc.close();
    }
}



// Q17. Accept an integer and check whether it is a palindrome using arithmetic operations and type casting.

import java.util.Scanner;

public class Lab01_Q17 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q17: Palindrome Integer Check ---");
        System.out.print("Enter an integer: ");
        int n = sc.nextInt();
        int original = n;
        int reversed = 0;
        while (n > 0) {
            int digit = n % 10;
            reversed = reversed * 10 + digit;
            n /= 10;
        }
        if (original == reversed) {
            System.out.println(original + " is a Palindrome number.");
        } else {
            System.out.println(original + " is NOT a Palindrome number.");
        }
        sc.close();
    }
}

// Q24. Accept a number and check whether it is a perfect number.

import java.util.Scanner;

public class Lab01_Q24 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q24: Perfect Number Check ---");
        System.out.print("Enter a positive integer: ");
        int n = sc.nextInt();
        if (n <= 0) {
            System.out.println("Please enter a positive integer.");
        } else {
            int sum = 0;
            for (int i = 1; i <= n / 2; i++) {
                if (n % i == 0) {
                    sum += i;
                }
            }
            if (sum == n) {
                System.out.println(n + " is a Perfect Number.");
            } else {
                System.out.println(n + " is NOT a Perfect Number.");
            }
        }
        sc.close();
    }
}
