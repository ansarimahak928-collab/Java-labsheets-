// Q1. Area of Triangle using Heron's Formula

import java.util.Scanner;

public class Lab02_Q01 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q1: Area of Triangle using Heron's Formula ---");
        System.out.print("Enter side a: ");
        double a = sc.nextDouble();
        System.out.print("Enter side b: ");
        double b = sc.nextDouble();
        System.out.print("Enter side c: ");
        double c = sc.nextDouble();

        if ((a + b > c) && (a + c > b) && (b + c > a)) {
            double s = (a + b + c) / 2.0;
            double area = Math.sqrt(s * (s - a) * (s - b) * (s - c));
            System.out.println("Area of the triangle: " + area);
        } else {
            System.out.println("Invalid triangle sides!");
        }
        sc.close();
    }
}

// Q9. Lexicographic String Comparison without built-ins

import java.util.Scanner;

public class Lab02_Q09 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q9: Lexicographic String Comparison without built-ins ---");
        System.out.print("Enter first string: ");
        String s1 = sc.next();
        System.out.print("Enter second string: ");
        String s2 = sc.next();

        int len1 = s1.length();
        int len2 = s2.length();
        int minLen = Math.min(len1, len2);
        boolean compared = false;

        for (int i = 0; i < minLen; i++) {
            char c1 = s1.charAt(i);
            char c2 = s2.charAt(i);
            if (c1 != c2) {
                if (c1 < c2) {
                    System.out.println("'" + s1 + "' comes before '" + s2 + "' lexicographically.");
                } else {
                    System.out.println("'" + s2 + "' comes before '" + s1 + "' lexicographically.");
                }
                compared = true;
                break;
            }
        }
        if (!compared) {
            if (len1 < len2) {
                System.out.println("'" + s1 + "' comes before '" + s2 + "' lexicographically.");
            } else if (len1 > len2) {
                System.out.println("'" + s2 + "' comes before '" + s1 + "' lexicographically.");
            } else {
                System.out.println("Both strings are identical.");
            }
        }
        sc.close();
    }
}

// Q15. Count Set Bits (1s) in Binary Representation

// Q19. Armstrong Numbers Between 1 and 1000

import java.util.Scanner;

public class Lab02_Q19 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q19: Armstrong Numbers Between 1 and 1000 ---");
        System.out.print("Armstrong numbers: ");
        for (int i = 1; i <= 1000; i++) {
            int sum = 0, digits = 0;
            int t = i;
            while (t > 0) { digits++; t /= 10; }
            t = i;
            while (t > 0) {
                int d = t % 10;
                sum += Math.pow(d, digits);
                t /= 10;
            }
            if (sum == i) {
                System.out.print(i + " ");
            }
        }
        System.out.println();
        sc.close();
    }
}
