// Q13. Multiply two matrices (3x3 matrices case)

import java.util.Scanner;

public class Lab03_Q13 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q13: Multiply two 3x3 matrices ---");
        int[][] a = new int[3][3];
        int[][] b = new int[3][3];
        int[][] prod = new int[3][3];

        System.out.println("Enter matrix A (3x3):");
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                a[i][j] = sc.nextInt();
            }
        }
        System.out.println("Enter matrix B (3x3):");
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                b[i][j] = sc.nextInt();
            }
        }
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                prod[i][j] = 0;
                for (int k = 0; k < 3; k++) {
                    prod[i][j] += a[i][k] * b[k][j];
                }
            }
        }
        System.out.println("Product Matrix:");
        for (int i = 0; i < 3; i++) {
            for (int j = 0; j < 3; j++) {
                System.out.print(prod[i][j] + "\t");
            }
            System.out.println();
        }
        sc.close();
    }
}

// Q22. Count how many times each element appears in a 1-D array

        System.out.println("--- Q22: Count frequency of each element ---");
        System.out.print("Enter size: ");
        int n = sc.nextInt();
        int[] arr = new int[n];
        boolean[] visited = new boolean[n];
        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        for (int i = 0; i < n; i++) {
            if (visited[i]) continue;
            int count = 1;
            for (int j = i + 1; j < n; j++) {
                if (arr[i] == arr[j]) {
                    visited[j] = true;
                    count++;
                }
            }
            System.out.println(arr[i] + " appears " + count + " times.");
        }
        sc.close();
    }
}

        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int first = Integer.MIN_VALUE, second = Integer.MIN_VALUE;
        for (int val : arr) {
            if (val > first) {
                second = first;
                first = val;
            } else if (val > second && val != first) {
                second = val;
            }
        }
        if (second == Integer.MIN_VALUE) {
            System.out.println("No second largest element found.");
        } else {
            System.out.println("Second largest element: " + second);
        }
        sc.close();
    }

// Q24. Remove duplicate elements from a 1-D array (without collections)

import java.util.Scanner;

public class Lab03_Q24 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q24: Remove duplicates without collections ---");
        System.out.print("Enter size: ");
        int n = sc.nextInt();
        int[] arr = new int[n];
        System.out.println("Enter elements:");
        for (int i = 0; i < n; i++) arr[i] = sc.nextInt();

        int newSize = n;
        for (int i = 0; i < newSize; i++) {
            for (int j = i + 1; j < newSize; j++) {
                if (arr[i] == arr[j]) {
                    for (int k = j; k < newSize - 1; k++) {
                        arr[k] = arr[k + 1];
                    }
                    newSize--;
                    j--;
                }
            }
        }
        System.out.print("Array after removing duplicates: ");
        for (int i = 0; i < newSize; i++) {
            System.out.print(arr[i] + " ");
        }
        System.out.println();
        sc.close();
    }
}



// Q25. Find the sum of main diagonal and secondary diagonal elements of a square matrix

        System.out.println("--- Q25: Main and secondary diagonal sum ---");
        System.out.print("Enter matrix order N: ");
        int n = sc.nextInt();
        int[][] mat = new int[n][n];
        int mainSum = 0, secSum = 0;

        System.out.println("Enter matrix elements:");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                mat[i][j] = sc.nextInt();
                if (i == j) mainSum += mat[i][j];
                if (i + j == n - 1) secSum += mat[i][j];
            }
        }
        System.out.println("Main diagonal sum: " + mainSum);
        System.out.println("Secondary diagonal sum: " + secSum);
        sc.close();
    }
}

                else System.out.print("0\t");
            }
            System.out.println();
        }
        System.out.println("Lower Triangular:");
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (j <= i) System.out.print(mat[i][j] + "\t");
                else System.out.print("0\t");
            }
            System.out.println();
        }
        sc.close();
    }
}

// Q30. Check if a given matrix is a sparse matrix (more zero elements than non-zero)

import java.util.Scanner;

public class Lab03_Q30 {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("--- Q30: Check if matrix is sparse matrix ---");
        System.out.print("Enter rows and cols: ");
        int r = sc.nextInt();
        int c = sc.nextInt();
        int[][] mat = new int[r][c];
        int zeros = 0, nonZeros = 0;

        System.out.println("Enter matrix elements:");
        for (int i = 0; i < r; i++) {
            for (int j = 0; j < c; j++) {
                mat[i][j] = sc.nextInt();
                if (mat[i][j] == 0) zeros++;
                else nonZeros++;
            }
        }
        if (zeros > nonZeros) {
            System.out.println("The matrix IS a sparse matrix (Zeros: " + zeros + ", Non-zeros: " + nonZeros + ").");
        } else {
            System.out.println("The matrix is NOT a sparse matrix (Zeros: " + zeros + ", Non-zeros: " + nonZeros + ").");
        }
        sc.close();
    }
}
