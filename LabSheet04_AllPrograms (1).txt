    public TriangleQ16(double a, double b, double c) {
        this.a = a;
        this.b = b;
        this.c = c;
    }

    public void checkValidityAndType() {
        if ((a + b > c) && (a + c > b) && (b + c > a)) {
            String triType = "";
            if (a == b && b == c) triType = "Equilateral";
            else if (a == b || b == c || a == c) triType = "Isosceles";
            else triType = "Scalene";
            System.out.println("Valid Triangle! Type: " + triType);
        } else {
            System.out.println("Invalid Triangle sides.");
        }
    }

    static int countChecks = 0;

    public void checkPalindrome(int num) {
        countChecks++;
        int original = num, rev = 0;
        while (num > 0) {
            rev = rev * 10 + (num % 10);
            num /= 10;
        }
        boolean isPal = (original == rev);
        System.out.println("Number " + original + " is palindrome: " + isPal + " | Total Checks: " + countChecks);
    }

// Q22. Class Armstrong check with class variable totalChecks

import java.util.Scanner;

class ArmstrongChecker {
    static int totalChecks = 0;

    public void checkArmstrong(int num) {
        totalChecks++;
        int original = num, sum = 0, digits = 0;
        int t = num;
        while (t > 0) { digits++; t /= 10; }
        t = num;
        while (t > 0) {
            int d = t % 10;
            sum += Math.pow(d, digits);
            t /= 10;
        }
        boolean isArm = (sum == original);
        System.out.println("Number " + original + " is Armstrong: " + isArm + " | Total Checks: " + totalChecks);
    }
}

public class Lab04_Q22 {
    public static void main(String[] args) {
        System.out.println("--- Q22: Armstrong Checker ---");
        ArmstrongChecker ac = new ArmstrongChecker();
        ac.checkArmstrong(153);
        ac.checkArmstrong(100);
    }
}



// Q23. Class PrimeChecker with class variable totalPrimeChecks

import java.util.Scanner;

class PrimeChecker {
    static int totalPrimeChecks = 0;

    public void checkPrime(int num) {
        totalPrimeChecks++;
        boolean isPrime = true;
        if (num <= 1) isPrime = false;
        for (int i = 2; i <= Math.sqrt(num); i++) {
            if (num % i == 0) {
                isPrime = false;
                break;
            }
        }
        System.out.println("Number " + num + " is prime: " + isPrime + " | Total Prime Checks: " + totalPrimeChecks);
    }
}

// Q30. Class Library with booksAvailable and class variable libraryName

class Library {
    int booksAvailable;
    static String libraryName = "Central Library COER";

    public Library(int booksAvailable) {
        this.booksAvailable = booksAvailable;
    }

    public void issueBook() {
        if (booksAvailable > 0) {
            booksAvailable--;
            System.out.println("Library: " + libraryName + " | Book issued. Remaining books: " + booksAvailable);
        } else {
            System.out.println("No books available to issue!");
        }
    }

    public void returnBook() {
        booksAvailable++;
        System.out.println("Library: " + libraryName + " | Book returned. Available books: " + booksAvailable);
    }
}

public class Lab04_Q30 {
    public static void main(String[] args) {
        System.out.println("--- Q30: Library Issue and Return Operations ---");

        lib.issueBook();
        lib.returnBook();
    }
